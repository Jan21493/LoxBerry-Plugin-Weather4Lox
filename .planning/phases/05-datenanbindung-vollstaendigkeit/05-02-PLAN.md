// ===== CONFIGURATION =====
var CGI_BASE = '../../htmlauth/show.cgi';
var ICON_BASE = 'icons/color/';
var ICONSET = getUrlParam('iconset') || 'color';
var SUPPORTED_LANGS = ['de', 'en', 'es', 'nl', 'sk'];
var currentLang = (function() {
  var p = getUrlParam('lang');
  return (p && SUPPORTED_LANGS.indexOf(p) !== -1) ? p : 'de';
})();
var REFRESH_INTERVAL_MS = 5 * 60 * 1000;

// Global state (replaces MOCK_* variables)
var liveData = { current: null, hourlyToday: [], hourlyTomorrow: [], daily: [], lang: null, iconMapping: null };

// ===== FETCH HELPERS =====
function checkOk(response) {
  if (!response.ok) throw new Error('HTTP ' + response.status + ' bei ' + response.url);
  return response;
}

function showLoading() { document.getElementById('loadingOverlay').style.display = 'flex'; }
function hideLoading() { document.getElementById('loadingOverlay').style.display = 'none'; }
function showError(msg) {
  var b = document.getElementById('errorBanner');
  var m = document.getElementById('errorMsg');
  m.textContent = msg || (liveData.lang ? liveData.lang.theme.error_loading : 'Daten konnten nicht geladen werden.');
  b.style.display = 'block';
  setTimeout(function() { b.style.display = 'none'; }, 8000);
}

// ===== DATA LOADING =====
function loadAllData() {
  showLoading();
  return Promise.all([
    fetch(CGI_BASE + '?format=json&type=current').then(checkOk).then(function(r) { return r.json(); }),
    fetch(CGI_BASE + '?format=json&type=hourly').then(checkOk).then(function(r) { return r.json(); }),
    fetch(CGI_BASE + '?format=json&type=daily').then(checkOk).then(function(r) { return r.json(); }),
    fetch('lang-' + currentLang + '.json').then(checkOk).then(function(r) { return r.json(); }),
    fetch('icons/' + ICONSET + '/icon_mapping.json').then(checkOk).then(function(r) { return r.json(); })
  ]).then(function(results) {
    hideLoading();
    liveData.current = results[0].data;
    liveData.lang = results[3];
    liveData.iconMapping = results[4].icons;

    // Split hourly into today/tomorrow using datetime from current.json as reference
    var todayStr = liveData.current.datetime.slice(0, 10);
    var todayDate = new Date(todayStr + 'T00:00:00');
    var tomorrowDate = new Date(todayDate); tomorrowDate.setDate(todayDate.getDate() + 1);
    var pad2 = function(n) { return n < 10 ? '0' + n : '' + n; };
    var tomorrowStr = tomorrowDate.getFullYear() + '-' + pad2(tomorrowDate.getMonth() + 1) + '-' + pad2(tomorrowDate.getDate());

    var hourlyRaw = results[1].data;
    liveData.hourlyToday = hourlyRaw.filter(function(h) { return h.datetime.slice(0, 10) === todayStr; });
    liveData.hourlyTomorrow = hourlyRaw.filter(function(h) { return h.datetime.slice(0, 10) === tomorrowStr; });

    // Transform daily data
    var dailyRaw = results[2].data;
    var weekdaysShort = liveData.lang.theme.weekdays_short;
    liveData.daily = dailyRaw.map(function(d) {
      var dt = new Date(d.datetime);
      return {
        day: weekdaysShort[dt.getDay()],
        date: pad2(dt.getDate()) + '.' + pad2(dt.getMonth() + 1) + '.',
        icon: d.weather_icon,
        high: d.high_temp,
        low: d.low_temp,
        pop: d.precip_probability,
        desc: d.weather_description
      };
    });

    renderAll();
    return liveData;
  }).catch(function(err) {
    hideLoading();
    showError(err.message);
    throw err;
  });
}

// ===== ICON RESOLUTION via icon_mapping.json =====
function isDaytime(datetimeStr, sunriseStr, sunsetStr) {
  var h = parseInt(datetimeStr.slice(11, 13));
  var sr = parseInt(sunriseStr.split(':')[0]);
  var ss = parseInt(sunsetStr.split(':')[0]);
  return h >= sr && h < ss;
}

function resolveIcon(code, dn) {
  // If icon_mapping loaded, use it; otherwise fallback
  if (liveData.iconMapping) {
    var key = code.replace(/-/g, '_');
    var entry = liveData.iconMapping[key];
    if (entry) {
      var field = (dn === 'day') ? 'icon_day' : 'icon_night';
      return 'icons/' + ICONSET + '/' + entry[field];
    }
  }
  return 'icons/' + ICONSET + '/' + code + '_' + (dn || 'day') + '.png';
}

// ===== TRANSFORM HOURLY for renderHourStrip =====
function transformHourly(hourlyArray, sunriseStr, sunsetStr) {
  return hourlyArray.map(function(h) {
    var hourStr = h.datetime.slice(11, 13);
    var dn = isDaytime(h.datetime, sunriseStr, sunsetStr) ? 'day' : 'night';
    return {
      hour: hourStr,
      temp: h.temperature,
      pop: h.precip_probability,
      wind: h.wind_speed,
      wind_dir: h.wind_direction_desc,
      humidity: h.humidity,
      icon: h.weather_icon,
      dn: dn,
      desc: h.weather_description
    };
  });
}

// ===== i18n =====
function applyLang(strings) {
  document.querySelectorAll('[data-i18n]').forEach(function(el) {
    var key = el.getAttribute('data-i18n');
    var parts = key.split('.');
    var val = strings;
    for (var i = 0; i < parts.length; i++) {
      val = val && val[parts[i]];
    }
    if (val && typeof val === 'string') el.textContent = val;
  });
}
