document.querySelectorAll('.tab').forEach(function(btn) {
  btn.addEventListener('click', function() {

        preselectHour(tabName);
      }, 300);

      document.querySelectorAll('.tab').forEach(function(b) { b.classList.remove('active'); });

**5. selectHour() function (adapted from ocean.main.html lines 591-606):**
```javascript
function selectHour(el, dataArray) {
  document.querySelectorAll('.hcard').forEach(function(c) { c.classList.remove('selected'); });
  el.classList.add('selected');
  var idx = parseInt(el.getAttribute('data-idx'));
  var d = dataArray[idx];
  var heroTemp = document.getElementById('heroTemp');
  heroTemp.style.opacity = '0';
  setTimeout(function() {
    heroTemp.innerHTML = Math.round(d.temp) + '<sup>\u00b0C</sup>';
    heroTemp.style.opacity = '1';
  }, 150);
  document.getElementById('heroDesc').textContent = d.desc;
  document.getElementById('heroWind').textContent = d.wind + ' km/h ' + d.wind_dir;
  document.getElementById('heroHumidity').textContent = d.humidity + '%';
  document.getElementById('heroIcon').src = ICON_BASE + d.icon + '_' + d.dn + '.png';
}
```

**6. preselectHour() function (HERO-06):**
```javascript
function preselectHour(tabName) {
  if (tabName === 'tage') return;
  var stripId = (tabName === 'heute') ? 'hourStripHeute' : 'hourStripMorgen';
  var dataArray = (tabName === 'heute') ? MOCK_HOURLY_TODAY : MOCK_HOURLY_TOMORROW;
  var cards = document.getElementById(stripId).querySelectorAll('.hcard');
  var targetIdx = 0;
  if (tabName === 'heute') {
    var currentHour = new Date().getHours();
    targetIdx = Math.min(currentHour, cards.length - 1);
  }
  if (cards[targetIdx]) {
    selectHour(cards[targetIdx], dataArray);
    cards[targetIdx].scrollIntoView({ behavior: 'smooth', block: 'nearest', inline: 'center' });
  }
}
```
